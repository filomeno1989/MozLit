    15→

    29→};

    59→  capa_url: string;

    62→  sinopse: string;
    63→  categorias: string[];
    64→  ficha_tecnica: string;

    70→
    71→export default function AuthorDashboard() {
    72→  const { user, navigate, token } = useAppStore();
    73→  const [data, setData] = useState<DashboardData | null>(null);
    74→  const [loading, setLoading] = useState(true);
    75→  const [bookChapters, setBookChapters] = useState<BookWithChapters | null>(null);
    76→  const [showChapterDialog, setShowChapterDialog] = useState(false);
    77→  const [newChapter, setNewChapter] = useState({ titulo: '', conteudo: '', preco_capitulo: '0', is_free: false });
    78→
    79→  // Edit book dialog
    80→  const [editingBook, setEditingBook] = useState<BookWithChapters | null>(null);
    81→  const [editForm, setEditForm] = useState({
    82→    titulo: '', sinopse: '', capa_url: '', preco_total: '',
    83→    categorias: [] as string[], categoriaInput: '',
    84→    ficha_tecnica: '', dedicatoria: '', epigrafe: '', epilogo: '',
    85→  });
    86→  const [editShowCatSugg, setEditShowCatSugg] = useState(false);
    87→  const [editActiveSections, setEditActiveSections] = useState<SectionKey[]>([]);
    88→  const [editUploading, setEditUploading] = useState(false);
    89→  const editFileRef = useRef<HTMLInputElement>(null);
    90→
    91→  // Profile dialog
    92→  const [showProfileDialog, setShowProfileDialog] = useState(false);
    93→  const [profileForm, setProfileForm] = useState({ biografia: '', avatar_url: '' });
    94→  const [profileUploading, setProfileUploading] = useState(false);
    95→  const profileFileRef = useRef<HTMLInputElement>(null);
    96→
    97→  useEffect(() => {
    98→    if (token) loadDashboard();
    99→  }, [token]);
   100→
   101→  async function loadDashboard() {
   102→    setLoading(true);
   103→    try {
   104→      const d = await apiFetch<DashboardData>('/api/author');
   105→      setData(d);
   106→      setProfileForm({ biografia: d.biografia ?? '', avatar_url: d.avatar_url ?? '' });
   107→    } catch {
   108→      setData(null);
   109→    } finally {
   110→      setLoading(false);
   111→    }
   112→  }
   113→
   114→  async function loadBookChapters(bookId: string) {
   115→    try {
   116→      const b = await apiFetch<BookWithChapters>(`/api/books/${bookId}`);
   117→      setBookChapters(b);
   118→      setShowChapterDialog(true);
   119→    } catch {
   120→      alert('Erro ao carregar capítulos');
   121→    }
   122→  }
   123→
   124→  async function publishBook(bookId: string) {
   125→    try {
   126→      await apiFetch(`/api/books/${bookId}`, { method: 'PATCH', body: JSON.stringify({ status: 'PUBLICADO' }) });
   127→      loadDashboard();
   128→    } catch (err) { alert((err as Error).message); }
   129→  }
   130→
   131→  async function deleteBook(bookId: string) {
   132→    if (!confirm('Tem certeza que deseja excluir este livro?')) return;
   133→    try {
   134→      await apiFetch(`/api/books/${bookId}`, { method: 'DELETE' });
   135→      loadDashboard();
   136→    } catch (err) { alert((err as Error).message); }
   137→  }
   138→
   139→  async function addChapter() {
   140→    if (!bookChapters || !newChapter.titulo || !newChapter.conteudo) {
   141→      alert('Título e conteúdo são obrigatórios.'); return;
   142→    }
   143→    try {
   144→      await apiFetch('/api/chapters', {
   145→        method: 'POST',
   146→        body: JSON.stringify({
   147→          titulo: newChapter.titulo, conteudo: newChapter.conteudo,
   148→          livroId: bookChapters.id, preco_capitulo: parseFloat(newChapter.preco_capitulo) || 0,
   149→          is_free: newChapter.is_free,
   150→        }),
   151→      });
   152→      setNewChapter({ titulo: '', conteudo: '', preco_capitulo: '0', is_free: false });
   153→      loadBookChapters(bookChapters.id);
   154→      loadDashboard();
   155→    } catch (err) { alert((err as Error).message); }
   156→  }
   157→
   158→  // --- Upload helper ---
   159→  async function uploadFile(file: File): Promise<string> {
   160→    const allowed = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
   161→    if (!allowed.includes(file.type)) throw new Error(`Tipo não suportado (${file.type}). Use JPG, PNG, WEBP ou GIF.`);
   162→    if (file.size > 5 * 1024 * 1024) throw new Error(`Ficheiro demasiado grande (${(file.size / 1024 / 1024).toFixed(1)}MB). Máximo 5MB.`);
   163→
   164→    const formData = new FormData();
   165→    formData.append('file', file);
   166→    const res = await fetch('/api/upload', {
   167→      method: 'POST', headers: { Authorization: `Bearer ${token}` }, body: formData,
   168→    });
   169→    if (res.status === 429) throw new Error('Muitas requisições. Aguarde e tente novamente.');
   170→    if (res.status === 502) throw new Error('Servidor indisponível. Recarregue a página.');
   171→    if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Erro no upload'); }
   172→    const d = await res.json();
   173→    return d.url;
   174→  }
   175→
   176→  // --- Edit Book ---
   177→  function openEditBook(livro: DashboardData['livros'][0]) {
   178→    apiFetch<BookWithChapters>(`/api/books/${livro.id}`).then((b) => {
   179→      setEditingBook(b);
   180→      // Determine which sections are active (have content)
   181→      const active: SectionKey[] = [];
   182→      if (b.ficha_tecnica) active.push('ficha_tecnica');
   183→      if (b.dedicatoria) active.push('dedicatoria');
   184→      if (b.epigrafe) active.push('epigrafe');
   185→      if (b.epilogo) active.push('epilogo');
   186→      setEditActiveSections(active);
   187→      setEditForm({
   188→        titulo: b.titulo, sinopse: b.sinopse,
   189→        capa_url: b.capa_url || '', preco_total: b.preco_total?.toString() ?? '0',
   190→        categorias: Array.isArray(b.categorias) ? b.categorias : [],
   191→        categoriaInput: '',
   192→        ficha_tecnica: b.ficha_tecnica || '',
   193→        dedicatoria: b.dedicatoria || '',
   194→        epigrafe: b.epigrafe || '',
   195→        epilogo: b.epilogo || '',
   196→      });
   197→    }).catch(() => alert('Erro ao carregar dados do livro'));
   198→  }
   199→
   200→  // Multi-categoria helpers for edit
   201→  const editCatFiltered = CATEGORIAS_SUGESTOES.filter(
   202→    (s) =>
   203→      s.toLowerCase().includes(editForm.categoriaInput.toLowerCase()) &&
   204→      !editForm.categorias.some((c) => c.toLowerCase() === s.toLowerCase())
   205→  );
   206→
   207→  function editAddCategoria(cat: string) {
   208→    const trimmed = cat.trim();
   209→    if (!trimmed) return;
   210→    if (editForm.categorias.some((c) => c.toLowerCase() === trimmed.toLowerCase())) return;
   211→    setEditForm({ ...editForm, categorias: [...editForm.categorias, trimmed], categoriaInput: '' });
   212→    setEditShowCatSugg(false);
   213→  }
   214→
   215→  function editRemoveCategoria(cat: string) {
   216→    setEditForm({ ...editForm, categorias: editForm.categorias.filter((c) => c !== cat) });
   217→  }
   218→
   219→  function editToggleSection(key: SectionKey) {
   220→    setEditActiveSections((prev) =>
   221→      prev.includes(key) ? prev.filter((s) => s !== key) : [...prev, key]
   222→    );
   223→  }
   224→
   225→  async function handleEditCoverUpload(e: React.ChangeEvent<HTMLInputElement>) {
   226→    const file = e.target.files?.[0];
   227→    if (!file) return;
   228→    setEditUploading(true);
   229→    try {
   230→      const url = await uploadFile(file);
   231→      setEditForm((f) => ({ ...f, capa_url: url }));
   232→    } catch (err) { alert((err as Error).message); }
   233→    finally { setEditUploading(false); if (editFileRef.current) editFileRef.current.value = ''; }
   234→  }
   235→
   236→  async function handleProfileAvatarUpload(e: React.ChangeEvent<HTMLInputElement>) {
   237→    const file = e.target.files?.[0];
   238→    if (!file) return;
   239→    setProfileUploading(true);
   240→    try {
   241→      const url = await uploadFile(file);
   242→      setProfileForm((f) => ({ ...f, avatar_url: url }));
   243→    } catch (err) { alert((err as Error).message); }
   244→    finally { setProfileUploading(false); if (profileFileRef.current) profileFileRef.current.value = ''; }
   245→  }
   246→
   247→  async function saveEditBook() {
   248→    if (!editingBook) return;
   249→    try {
   250→      const payload: Record<string, unknown> = {
   251→        titulo: editForm.titulo,
   252→        sinopse: editForm.sinopse,
   253→        capa_url: editForm.capa_url || undefined,
   254→        preco_total: parseFloat(editForm.preco_total) || 0,
   255→        categorias: editForm.categorias,
   256→      };
   257→      // Always send section fields (empty string to clear)
   258→      payload.ficha_tecnica = editForm.ficha_tecnica;
   259→      payload.dedicatoria = editForm.dedicatoria;
   260→      payload.epigrafe = editForm.epigrafe;
   261→      payload.epilogo = editForm.epilogo;
   262→
   263→      await apiFetch(`/api/books/${editingBook.id}`, { method: 'PATCH', body: JSON.stringify(payload) });
   264→      setEditingBook(null);
   265→      loadDashboard();
   266→    } catch (err) { alert((err as Error).message); }
   267→  }
   268→
   269→  async function saveProfile() {
   270→    try {
   271→      await apiFetch('/api/author/profile', { method: 'PATCH', body: JSON.stringify(profileForm) });
   272→      setShowProfileDialog(false);
   273→      loadDashboard();
   274→    } catch (err) { alert((err as Error).message); }
   275→  }
   276→
   277→  if (loading) {
   278→    return (
   279→      <div className="max-w-5xl mx-auto px-4 py-8 space-y-6">
   280→        <Skeleton className="h-8 w-64" />
   281→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   282→          <Skeleton className="h-24" /><Skeleton className="h-24" /><Skeleton className="h-24" />
   283→        </div>
   284→      </div>
   285→    );
   286→  }
   287→
   288→  return (
   289→    <div className="max-w-5xl mx-auto px-4 py-8">
   290→      <div className="flex items-center justify-between mb-8">
   291→        <div>
   292→          <h1 className="text-2xl font-bold">Painel do Autor</h1>
   293→          <p className="text-sm text-muted-foreground">Gerencie suas obras e acompanhe seus ganhos</p>
   294→        </div>
   295→        <div className="flex items-center gap-2">
   296→          <Button variant="outline" size="sm" onClick={() => setShowProfileDialog(true)}>
   297→            <User className="h-4 w-4 mr-1" /> Meu Perfil
   298→          </Button>
   299→          <Button className="bg-amber-600 hover:bg-amber-700 text-white" onClick={() => navigate('new-book')}>
   300→            <Plus className="h-4 w-4 mr-1" /> Nova Obra
   301→          </Button>
   302→        </div>
   303→      </div>
   304→
   305→      {/* Stats */}
   306→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-8">
   307→        <Card><CardContent className="p-4 flex items-center gap-3">
   308→          <div className="p-2 rounded-lg bg-amber-100 dark:bg-amber-900/30"><DollarSign className="h-5 w-5 text-amber-700 dark:text-amber-400" /></div>
   309→          <div><p className="text-xs text-muted-foreground">Saldo Total</p><p className="text-xl font-bold">{(data?.totalGanhos ?? 0).toFixed(2)} MZN</p></div>
   310→        </CardContent></Card>
   311→        <Card><CardContent className="p-4 flex items-center gap-3">
   312→          <div className="p-2 rounded-lg bg-emerald-100 dark:bg-emerald-900/30"><BookOpen className="h-5 w-5 text-emerald-700 dark:text-emerald-400" /></div>
   313→          <div><p className="text-xs text-muted-foreground">Obras</p><p className="text-xl font-bold">{data?.totalLivros ?? 0}</p></div>
   314→        </CardContent></Card>
   315→        <Card><CardContent className="p-4 flex items-center gap-3">
   316→          <div className="p-2 rounded-lg bg-blue-100 dark:bg-blue-900/30"><FileText className="h-5 w-5 text-blue-700 dark:text-blue-400" /></div>
   317→          <div><p className="text-xs text-muted-foreground">Capítulos</p><p className="text-xl font-bold">{data?.totalCapitulos ?? 0}</p></div>
   318→        </CardContent></Card>
   319→      </div>
   320→
   321→      {/* Livros */}
   322→      <Card>
   323→        <CardHeader><CardTitle className="text-lg">Minhas Obras</CardTitle></CardHeader>
   324→        <CardContent>
   325→          {!data?.livros.length ? (
   326→            <p className="text-sm text-muted-foreground py-4">Nenhuma obra criada ainda.</p>
   327→          ) : (
   328→            <div className="space-y-3">
   329→              {data.livros.map((livro) => (
   330→                <div key={livro.id} className="flex flex-col sm:flex-row sm:items-center justify-between p-3 rounded-lg border border-border/50 gap-2">
   331→                  <div className="flex items-center gap-3 min-w-0">
   332→                    {livro.capa_url && livro.capa_url !== '/placeholder-cover.svg' ? (
   333→                      <img src={livro.capa_url} alt={livro.titulo} className="w-10 h-14 rounded object-cover shrink-0 border border-border/30" />
   334→                    ) : (
   335→                      <div className="w-10 h-14 rounded bg-muted flex items-center justify-center shrink-0"><BookOpen className="h-4 w-4 text-muted-foreground" /></div>
   336→                    )}
   337→                    <div className="min-w-0">
   338→                      <div className="flex items-center gap-2">
   339→                        <h3 className="font-medium text-sm truncate">{livro.titulo}</h3>
   340→                        <Badge variant={livro.status === 'PUBLICADO' ? 'default' : 'secondary'} className="text-xs shrink-0">
   341→                          {livro.status === 'PUBLICADO' ? 'Publicado' : 'Rascunho'}
   342→                        </Badge>
   343→                      </div>
   344→                      <p className="text-xs text-muted-foreground mt-0.5">
   345→                        {livro.categorias.join(', ')} - {livro.totalCapitulos} cap. ({livro.capitulosPagos} pagos)
   346→                        {livro.preco_total > 0 && ` - Completo: ${livro.preco_total.toFixed(2)} MZN`}
   347→                        {' - '}Receita: {livro.receitaEstimada.toFixed(2)} MZN
   348→                      </p>
   349→                    </div>
   350→                  </div>
   351→                  <div className="flex items-center gap-1.5 shrink-0 sm:ml-2">
   352→                    <Button size="sm" variant="ghost" onClick={() => loadBookChapters(livro.id)}><FileText className="h-3.5 w-3.5 mr-1" /> Capítulos</Button>
   353→                    <Button size="sm" variant="ghost" onClick={() => openEditBook(livro)}><Pencil className="h-3.5 w-3.5" /></Button>
   354→                    {livro.status !== 'PUBLICADO' && (
   355→                      <Button size="sm" variant="outline" className="text-emerald-600" onClick={() => publishBook(livro.id)}><Eye className="h-3.5 w-3.5 mr-1" /> Publicar</Button>
   356→                    )}
   357→                    <Button size="sm" variant="ghost" className="text-destructive" onClick={() => deleteBook(livro.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   358→                  </div>
   359→                </div>
   360→              ))}
   361→            </div>
   362→          )}
   363→        </CardContent>
   364→      </Card>
   365→
   366→      {/* Chapter Dialog */}
   367→      <Dialog open={showChapterDialog} onOpenChange={setShowChapterDialog}>
   368→        <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto">
   369→          <DialogHeader><DialogTitle>Capítulos - {bookChapters?.titulo}</DialogTitle></DialogHeader>
   370→          {bookChapters && (
   371→            <div className="space-y-4">
   372→              <div className="space-y-1.5 max-h-40 overflow-y-auto">
   373→                {bookChapters.chapters.map((ch) => (
   374→                  <div key={ch.id} className="flex items-center justify-between text-sm p-2 rounded bg-muted/50">
   375→                    <span className="truncate">
   376→                      <span className="text-muted-foreground font-mono text-xs mr-2">{String(ch.ordem + 1).padStart(2, '0')}</span>
   377→                      {ch.titulo}
   378→                      {ch.is_free && <Badge variant="secondary" className="ml-2 text-xs">Grátis</Badge>}
   379→                    </span>
   380→                    <span className="text-xs text-muted-foreground shrink-0 ml-2">{ch.is_free ? 'Grátis' : `${ch.preco_capitulo.toFixed(2)} MZN`}</span>
   381→                  </div>
   382→                ))}
   383→                {bookChapters.chapters.length === 0 && <p className="text-sm text-muted-foreground py-2">Nenhum capítulo.</p>}
   384→              </div>
   385→              <div className="border-t pt-4 space-y-3">
   386→                <p className="text-sm font-medium">Adicionar Capítulo</p>
   387→                <div><Label className="text-xs">Título</Label><Input value={newChapter.titulo} onChange={(e) => setNewChapter({ ...newChapter, titulo: e.target.value })} placeholder="Título do capítulo" /></div>
   388→                <div><Label className="text-xs">Conteúdo</Label><Textarea value={newChapter.conteudo} onChange={(e) => setNewChapter({ ...newChapter, conteudo: e.target.value })} placeholder="Escreva o conteúdo do capítulo..." rows={6} /></div>
   389→                <div className="flex items-center gap-4">
   390→                  <div className="flex-1"><Label className="text-xs">Preço (MZN)</Label><Input type="number" step="0.01" min="0" value={newChapter.preco_capitulo} onChange={(e) => setNewChapter({ ...newChapter, preco_capitulo: e.target.value })} disabled={newChapter.is_free} /></div>
   391→                  <div className="flex items-center gap-2 pt-5">
   392→                    <input type="checkbox" id="free-check" checked={newChapter.is_free} onChange={(e) => setNewChapter({ ...newChapter, is_free: e.target.checked })} className="rounded" />
   393→                    <Label htmlFor="free-check" className="text-xs">Grátis</Label>
   394→                  </div>
   395→                </div>
   396→                <Button onClick={addChapter} className="w-full bg-amber-600 hover:bg-amber-700 text-white"><Plus className="h-4 w-4 mr-1" /> Adicionar Capítulo</Button>
   397→              </div>
   398→            </div>
   399→          )}
   400→        </DialogContent>
   401→      </Dialog>
   402→
   403→      {/* Edit Book Dialog */}
   404→      <Dialog open={!!editingBook} onOpenChange={(open) => !open && setEditingBook(null)}>
   405→        <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto">
   406→          <DialogHeader><DialogTitle>Editar Obra</DialogTitle></DialogHeader>
   407→          {editingBook && (
   408→            <div className="space-y-4">
   409→              <div><Label>Título</Label><Input value={editForm.titulo} onChange={(e) => setEditForm({ ...editForm, titulo: e.target.value })} /></div>
   410→              <div><Label>Sinopse</Label><Textarea value={editForm.sinopse} onChange={(e) => setEditForm({ ...editForm, sinopse: e.target.value })} rows={3} /></div>
   411→
   412→              {/* Multi-categoria */}
   413→              <div>
   414→                <Label>Categorias</Label>
   415→                <div className="relative mt-1.5">
   416→                  <Input
   417→                    value={editForm.categoriaInput}
   418→                    onChange={(e) => { setEditForm({ ...editForm, categoriaInput: e.target.value }); setEditShowCatSugg(true); }}
   419→                    onFocus={() => setEditShowCatSugg(true)}
