   161→   161→  // --- Upload helper ---
   162→   162→  async function uploadFile(file: File): Promise<string> {
   163→   163→    const allowed = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
   164→   164→    if (!allowed.includes(file.type)) throw new Error(`Tipo não suportado (${file.type}). Use JPG, PNG, WEBP ou GIF.`);
   165→   165→    if (file.size > 5 * 1024 * 1024) throw new Error(`Ficheiro demasiado grande (${(file.size / 1024 / 1024).toFixed(1)}MB). Máximo 5MB.`);
   166→   166→
   167→   167→    const formData = new FormData();
   168→   168→    formData.append('file', file);
   169→   169→    const res = await fetch('/api/upload', {
   170→   170→      method: 'POST', headers: { Authorization: `Bearer ${token}` }, body: formData,
   171→   171→    });
   172→   172→    if (res.status === 429) throw new Error('Muitas requisições. Aguarde e tente novamente.');
   173→   173→    if (res.status === 502) throw new Error('Servidor indisponível. Recarregue a página.');
   174→   174→    if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Erro no upload'); }
   175→   175→    const d = await res.json();
   176→   176→    return d.url;
   177→   177→  }
