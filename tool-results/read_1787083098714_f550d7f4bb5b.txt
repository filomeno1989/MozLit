    65→  categorias: string[];

    72→
    73→export default function AuthorDashboard() {
    74→  const { user, navigate, token } = useAppStore();
    75→  const [data, setData] = useState<DashboardData | null>(null);
    76→  const [loading, setLoading] = useState(true);
    77→  const [bookChapters, setBookChapters] = useState<BookWithChapters | null>(null);
    78→  const [showChapterDialog, setShowChapterDialog] = useState(false);
    79→  const [chapterError, setChapterError] = useState('');
    80→  const [newChapter, setNewChapter] = useState({ titulo: '', conteudo: '', preco_capitulo: '0', is_free: false });
    81→
    82→  // Edit book dialog
    83→  const [editingBook, setEditingBook] = useState<BookWithChapters | null>(null);
    84→  const [editForm, setEditForm] = useState({
    85→    titulo: '', sinopse: '', capa_url: '', preco_total: '',
    86→    categorias: [] as string[], categoriaInput: '',
    87→    ficha_tecnica: '', dedicatoria: '', epigrafe: '', epilogo: '',
    88→  });
    89→  const [editShowCatSugg, setEditShowCatSugg] = useState(false);
    90→  const [editActiveSections, setEditActiveSections] = useState<SectionKey[]>([]);
    91→  const [editUploading, setEditUploading] = useState(false);
    92→  const editFileRef = useRef<HTMLInputElement>(null);
    93→
    94→  // Delete confirmation dialog
    95→  const [deleteTarget, setDeleteTarget] = useState<string | null>(null);
    96→  const [deleting, setDeleting] = useState(false);
    97→
    98→  // Error state for dashboard
    99→  const [loadError, setLoadError] = useState(false);
   100→
   101→  // Profile dialog
   102→  const [showProfileDialog, setShowProfileDialog] = useState(false);
   103→  const [profileForm, setProfileForm] = useState({ biografia: '', avatar_url: '' });
   104→  const [profileUploading, setProfileUploading] = useState(false);
   105→  const profileFileRef = useRef<HTMLInputElement>(null);
   106→
   107→  useEffect(() => {
   108→    if (token) loadDashboard();
   109→  }, [token]);
   110→
   111→  async function loadDashboard() {
   112→    setLoading(true);
   113→    try {
   114→      const d = await apiFetch<DashboardData>('/api/author');
   115→      setData(d);
   116→      setProfileForm({ biografia: d.biografia ?? '', avatar_url: d.avatar_url ?? '' });
   117→    } catch {
   118→      setData(null);
   119→      setLoadError(true);
   120→    } finally {
   121→      setLoading(false);
   122→    }
   123→  }
   124→
   125→  async function loadBookChapters(bookId: string) {
   126→    try {
   127→      const b = await apiFetch<BookWithChapters>(`/api/books/${bookId}`);
   128→      setBookChapters(b);
   129→      setShowChapterDialog(true);
   130→    } catch (err) {
   131→      setChapterError((err as Error).message);
   132→    }
   133→  }
   134→
   135→  async function publishBook(bookId: string) {
   136→    try {
   137→      await apiFetch(`/api/books/${bookId}`, { method: 'PATCH', body: JSON.stringify({ status: 'PUBLICADO' }) });
   138→      loadDashboard();
   139→    } catch (err) { alert((err as Error).message); }
   140→  }
   141→
   142→  async function deleteBook() {
   143→    if (!deleteTarget) return;
   144→    setDeleting(true);
   145→    try {
   146→      await apiFetch(`/api/books/${deleteTarget}`, { method: 'DELETE' });
   147→      setDeleteTarget(null);
   148→      loadDashboard();
   149→    } catch (err) { alert((err as Error).message); }
   150→    finally { setDeleting(false); }
   151→  }
   152→
   153→  async function addChapter() {
   154→    if (!bookChapters || !newChapter.titulo || !newChapter.conteudo) {
   155→      setChapterError('Título e conteúdo são obrigatórios.'); return;
   156→    }
   157→    try {
   158→      await apiFetch('/api/chapters', {
   159→        method: 'POST',
   160→        body: JSON.stringify({
   161→          titulo: newChapter.titulo, conteudo: newChapter.conteudo,
   162→          livroId: bookChapters.id, preco_capitulo: parseFloat(newChapter.preco_capitulo) || 0,
   163→          is_free: newChapter.is_free,
   164→        }),
   165→      });
   166→      setNewChapter({ titulo: '', conteudo: '', preco_capitulo: '0', is_free: false });
   167→      setChapterError('');
   168→      loadBookChapters(bookChapters.id);
   169→      loadDashboard();
   170→    } catch (err) { setChapterError((err as Error).message); }
   171→  }
   172→
   173→  // --- Upload helper ---
   174→  async function uploadFile(file: File): Promise<string> {
   175→    const allowed = ['image/jpeg', 'image/png', 'image/webp', 'image/gif'];
   176→    if (!allowed.includes(file.type)) throw new Error(`Tipo não suportado (${file.type}). Use JPG, PNG, WEBP ou GIF.`);
   177→    if (file.size > 5 * 1024 * 1024) throw new Error(`Ficheiro demasiado grande (${(file.size / 1024 / 1024).toFixed(1)}MB). Máximo 5MB.`);
   178→
   179→    const formData = new FormData();
   180→    formData.append('file', file);
   181→    const res = await fetch('/api/upload', {
   182→      method: 'POST', headers: { Authorization: `Bearer ${token}` }, body: formData,
   183→    });
   184→    if (res.status === 429) throw new Error('Muitas requisições. Aguarde e tente novamente.');
   185→    if (res.status === 502) throw new Error('Servidor indisponível. Recarregue a página.');
   186→    if (!res.ok) { const d = await res.json(); throw new Error(d.error || 'Erro no upload'); }
   187→    const d = await res.json();
   188→    return d.url;
   189→  }
   190→
   191→  // --- Edit Book ---
   192→  function openEditBook(livro: DashboardData['livros'][0]) {
   193→    apiFetch<BookWithChapters>(`/api/books/${livro.id}`).then((b) => {
   194→      setEditingBook(b);
   195→      // Determine which sections are active (have content)
   196→      const active: SectionKey[] = [];
   197→      if (b.ficha_tecnica) active.push('ficha_tecnica');
   198→      if (b.dedicatoria) active.push('dedicatoria');
   199→      if (b.epigrafe) active.push('epigrafe');
   200→      if (b.epilogo) active.push('epilogo');
   201→      setEditActiveSections(active);
   202→      setEditForm({
   203→        titulo: b.titulo, sinopse: b.sinopse,
   204→        capa_url: b.capa_url || '', preco_total: b.preco_total?.toString() ?? '0',
   205→        categorias: Array.isArray(b.categorias) ? b.categorias : [],
   206→        categoriaInput: '',
   207→        ficha_tecnica: b.ficha_tecnica || '',
   208→        dedicatoria: b.dedicatoria || '',
   209→        epigrafe: b.epigrafe || '',
   210→        epilogo: b.epilogo || '',
   211→      });
   212→    }).catch(() => alert('Erro ao carregar dados do livro'));
   213→  }
   214→
   215→  // Multi-categoria helpers for edit
   216→  const editCatFiltered = CATEGORIAS_SUGESTOES.filter(
   217→    (s) =>
   218→      s.toLowerCase().includes(editForm.categoriaInput.toLowerCase()) &&
   219→      !editForm.categorias.some((c) => c.toLowerCase() === s.toLowerCase())
   220→  );
   221→
   222→  function editAddCategoria(cat: string) {
   223→    const trimmed = cat.trim();
   224→    if (!trimmed) return;
   225→    if (editForm.categorias.some((c) => c.toLowerCase() === trimmed.toLowerCase())) return;
   226→    setEditForm({ ...editForm, categorias: [...editForm.categorias, trimmed], categoriaInput: '' });
   227→    setEditShowCatSugg(false);
   228→  }
   229→
   230→  function editRemoveCategoria(cat: string) {
   231→    setEditForm({ ...editForm, categorias: editForm.categorias.filter((c) => c !== cat) });
   232→  }
   233→
   234→  function editToggleSection(key: SectionKey) {
   235→    setEditActiveSections((prev) =>
   236→      prev.includes(key) ? prev.filter((s) => s !== key) : [...prev, key]
   237→    );
   238→  }
   239→
   240→  async function handleEditCoverUpload(e: React.ChangeEvent<HTMLInputElement>) {
   241→    const file = e.target.files?.[0];
   242→    if (!file) return;
   243→    setEditUploading(true);
   244→    try {
   245→      const url = await uploadFile(file);
   246→      setEditForm((f) => ({ ...f, capa_url: url }));
   247→    } catch (err) { alert((err as Error).message); }
   248→    finally { setEditUploading(false); if (editFileRef.current) editFileRef.current.value = ''; }
   249→  }
   250→
   251→  async function handleProfileAvatarUpload(e: React.ChangeEvent<HTMLInputElement>) {
   252→    const file = e.target.files?.[0];
   253→    if (!file) return;
   254→    setProfileUploading(true);
   255→    try {
   256→      const url = await uploadFile(file);
   257→      setProfileForm((f) => ({ ...f, avatar_url: url }));
   258→    } catch (err) { alert((err as Error).message); }
   259→    finally { setProfileUploading(false); if (profileFileRef.current) profileFileRef.current.value = ''; }
   260→  }
   261→
   262→  async function saveEditBook() {
   263→    if (!editingBook) return;
   264→    try {
   265→      const payload: Record<string, unknown> = {
   266→        titulo: editForm.titulo,
   267→        sinopse: editForm.sinopse,
   268→        capa_url: editForm.capa_url || undefined,
   269→        preco_total: parseFloat(editForm.preco_total) || 0,
   270→        categorias: editForm.categorias,
   271→      };
   272→      // Always send section fields (empty string to clear)
   273→      payload.ficha_tecnica = editForm.ficha_tecnica;
   274→      payload.dedicatoria = editForm.dedicatoria;
   275→      payload.epigrafe = editForm.epigrafe;
   276→      payload.epilogo = editForm.epilogo;
   277→
   278→      await apiFetch(`/api/books/${editingBook.id}`, { method: 'PATCH', body: JSON.stringify(payload) });
   279→      setEditingBook(null);
   280→      loadDashboard();
   281→    } catch (err) { alert((err as Error).message); }
   282→  }
   283→
   284→  async function saveProfile() {
   285→    try {
   286→      await apiFetch('/api/author/profile', { method: 'PATCH', body: JSON.stringify(profileForm) });
   287→      setShowProfileDialog(false);
   288→      loadDashboard();
   289→    } catch (err) { alert((err as Error).message); }
   290→  }
   291→
   292→  if (loading) {
   293→    return (
   294→      <div className="max-w-5xl mx-auto px-4 py-8 space-y-6">
   295→        <Skeleton className="h-8 w-64" />
   296→        <div className="grid grid-cols-1 sm:grid-cols-3 gap-4">
   297→          <Skeleton className="h-24" /><Skeleton className="h-24" /><Skeleton className="h-24" />
   298→        </div>
   299→      </div>
   300→    );
   301→  }
   302→
   303→  return (
   304→    <div className="max-w-5xl mx-auto px-4 py-8">
   305→      {loadError ? (
   306→        <div className="text-center py-16">
   307→          <p className="text-destructive font-medium mb-2">Erro ao carregar o painel</p>
   308→          <p className="text-sm text-muted-foreground mb-4">Verifique sua conexão e tente novamente.</p>
   309→          <Button variant="outline" onClick={loadDashboard}>Tentar novamente</Button>
   310→        </div>
   311→      ) : (
   312→        <>
   313→        <div className="flex items-center justify-between mb-8">
   314→        <div>
   315→          <h1 className="text-2xl font-bold">Painel do Autor</h1>
   316→          <p className="text-sm text-muted-foreground">Gerencie suas obras e acompanhe seus ganhos</p>
   317→        </div>
   318→        <div className="flex items-center gap-2">
   319→          <Button variant="outline" size="sm" onClick={() => setShowProfileDialog(true)}>
   320→            <User className="h-4 w-4 mr-1" /> Meu Perfil
   321→          </Button>
   322→          <Button className="bg-amber-600 hover:bg-amber-700 text-white" onClick={() => navigate('new-book')}>
   323→            <Plus className="h-4 w-4 mr-1" /> Nova Obra
   324→          </Button>
   325→        </div>
   326→      </div>
   327→
   328→      {/* Stats */}
   329→      <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 mb-8">
   330→        <Card className="transition-shadow hover:shadow-md"><CardContent className="p-4 flex items-center gap-3">
   331→          <div className="p-2 rounded-lg bg-amber-100 dark:bg-amber-900/30"><DollarSign className="h-5 w-5 text-amber-700 dark:text-amber-400" /></div>
   332→          <div><p className="text-xs text-muted-foreground">Saldo Total</p><p className="text-xl font-bold">{(data?.totalGanhos ?? 0).toFixed(2)} MZN</p></div>
   333→        </CardContent></Card>
   334→        <Card className="transition-shadow hover:shadow-md"><CardContent className="p-4 flex items-center gap-3">
   335→          <div className="p-2 rounded-lg bg-emerald-100 dark:bg-emerald-900/30"><BookOpen className="h-5 w-5 text-emerald-700 dark:text-emerald-400" /></div>
   336→          <div><p className="text-xs text-muted-foreground">Obras</p><p className="text-xl font-bold">{data?.totalLivros ?? 0}</p></div>
   337→        </CardContent></Card>
   338→        <Card className="transition-shadow hover:shadow-md"><CardContent className="p-4 flex items-center gap-3">
   339→          <div className="p-2 rounded-lg bg-blue-100 dark:bg-blue-900/30"><FileText className="h-5 w-5 text-blue-700 dark:text-blue-400" /></div>
   340→          <div><p className="text-xs text-muted-foreground">Capítulos</p><p className="text-xl font-bold">{data?.totalCapitulos ?? 0}</p></div>
   341→        </CardContent></Card>
   342→      </div>
   343→
   344→      {/* Livros */}
   345→      <Card>
   346→        <CardHeader><CardTitle className="text-lg">Minhas Obras</CardTitle></CardHeader>
   347→        <CardContent>
   348→          {!data?.livros.length ? (
   349→            <p className="text-sm text-muted-foreground py-4">Nenhuma obra criada ainda.</p>
   350→          ) : (
   351→            <div className="space-y-3">
   352→              {data.livros.map((livro) => (
   353→                <div key={livro.id} className="flex flex-col sm:flex-row sm:items-center justify-between p-3 rounded-lg border border-border/50 gap-2">
   354→                  <div className="flex items-center gap-3 min-w-0">
   355→                    {livro.capa_url && livro.capa_url !== '/placeholder-cover.svg' ? (
   356→                      <img src={livro.capa_url} alt={livro.titulo} className="w-10 h-14 rounded object-cover shrink-0 border border-border/30" />
   357→                    ) : (
   358→                      <div className="w-10 h-14 rounded bg-muted flex items-center justify-center shrink-0"><BookOpen className="h-4 w-4 text-muted-foreground" /></div>
   359→                    )}
   360→                    <div className="min-w-0">
   361→                      <div className="flex items-center gap-2">
   362→                        <h3 className="font-medium text-sm truncate">{livro.titulo}</h3>
   363→                        <Badge variant={livro.status === 'PUBLICADO' ? 'default' : 'secondary'} className="text-xs shrink-0">
   364→                          {livro.status === 'PUBLICADO' ? 'Publicado' : 'Rascunho'}
   365→                        </Badge>
   366→                      </div>
   367→                      <p className="text-xs text-muted-foreground mt-0.5">
   368→                        {livro.categorias.join(', ')} - {livro.totalCapitulos} cap. ({livro.capitulosPagos} pagos)
   369→                        {livro.preco_total > 0 && ` - Completo: ${livro.preco_total.toFixed(2)} MZN`}
   370→                        {' - '}Receita: {livro.receitaEstimada.toFixed(2)} MZN
   371→                      </p>
   372→                    </div>
   373→                  </div>
   374→                  <div className="flex items-center gap-1.5 shrink-0 sm:ml-2">
   375→                    <Button size="sm" variant="ghost" onClick={() => loadBookChapters(livro.id)}><FileText className="h-3.5 w-3.5 mr-1" /> Capítulos</Button>
   376→                    <Button size="sm" variant="ghost" onClick={() => openEditBook(livro)}><Pencil className="h-3.5 w-3.5" /></Button>
   377→                    {livro.status !== 'PUBLICADO' && (
   378→                      <Button size="sm" variant="outline" className="text-emerald-600" onClick={() => publishBook(livro.id)}><Eye className="h-3.5 w-3.5 mr-1" /> Publicar</Button>
   379→                    )}
   380→                    <Button size="sm" variant="ghost" className="text-destructive" onClick={() => setDeleteTarget(livro.id)}><Trash2 className="h-3.5 w-3.5" /></Button>
   381→                  </div>
   382→                </div>
   383→              ))}
   384→            </div>
   385→          )}
   386→        </CardContent>
   387→      </Card>
   388→
   389→      {/* Chapter Dialog */}
   390→      <Dialog open={showChapterDialog} onOpenChange={setShowChapterDialog}>
   391→        <DialogContent className="max-w-lg max-h-[85vh] overflow-y-auto">
   392→          <DialogHeader><DialogTitle>Capítulos - {bookChapters?.titulo}</DialogTitle></DialogHeader>
   393→          {bookChapters && (
   394→            <div className="space-y-4">
   395→              {chapterError && (
   396→                <div className="p-2.5 rounded-lg bg-destructive/10 text-destructive text-sm">{chapterError}</div>
   397→              )}
   398→              <div className="divide-y divide-border/40">
   399→                {bookChapters.chapters.map((ch) => (
   400→                  <div key={ch.id} className="flex items-center justify-between text-sm p-2.5 first:pt-0 last:pb-0">
   401→                    <span className="truncate">
   402→                      <span className="text-muted-foreground font-mono text-xs mr-2">{String(ch.ordem + 1).padStart(2, '0')}</span>
   403→                      {ch.titulo}
   404→                      {ch.is_free && <Badge variant="secondary" className="ml-2 text-xs">Grátis</Badge>}
   405→                    </span>
   406→                    <span className="text-xs text-muted-foreground shrink-0 ml-2">{ch.is_free ? 'Grátis' : `${ch.preco_capitulo.toFixed(2)} MZN`}</span>
   407→                  </div>
   408→                ))}
   409→                {bookChapters.chapters.length === 0 && <p className="text-sm text-muted-foreground py-2">Nenhum capítulo.</p>}
   410→              </div>
   411→              <div className="border-t pt-4 space-y-3">
   412→                <p className="text-sm font-medium">Adicionar Capítulo</p>
   413→                <div><Label className="text-xs">Título</Label><Input value={newChapter.titulo} onChange={(e) => setNewChapter({ ...newChapter, titulo: e.target.value })} placeholder="Título do capítulo" /></div>
   414→                <div><Label className="text-xs">Conteúdo</Label><Textarea value={newChapter.conteudo} onChange={(e) => setNewChapter({ ...newChapter, conteudo: e.target.value })} placeholder="Escreva o conteúdo do capítulo..." rows={6} /></div>
   415→                <div className="flex items-center gap-4">
   416→                  <div className="flex-1"><Label className="text-xs">Preço (MZN)</Label><Input type="number" step="0.01" min="0" value={newChapter.preco_capitulo} onChange={(e) => setNewChapter({ ...newChapter, preco_capitulo: e.target.value })} disabled={newChapter.is_free} /></div>
   417→                  <div className="flex items-center gap-2 pt-5">
   418→                    <Checkbox id="free-check" checked={newChapter.is_free} onCheckedChange={(checked) => setNewChapter({ ...newChapter, is_free: !!checked })} />
   419→                    <Label htmlFor="free-check" className="text-xs cursor-pointer">Grátis</Label>
